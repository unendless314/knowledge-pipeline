# Knowledge Pipeline Cron Jobs
# 安裝方式: scripts/cron/install.sh
#
# 時間格式說明:
# ┌───────────── 分鐘 (0-59)
# │ ┌───────────── 小時 (0-23)
# │ │ ┌───────────── 日期 (1-31)
# │ │ │ ┌───────────── 月份 (1-12)
# │ │ │ │ ┌───────────── 星期 (0-7, 0=日, 7=日)
# │ │ │ │ │
# * * * * * 指令
#
# 常用範例:
# 0 3 * * *    -> 每天凌晨 3:00
# 0 9 * * *    -> 每天早上 9:00
# 0 */6 * * *  -> 每 6 小時

# ============================================
# Knowledge Pipeline - 每天凌晨 3:00 執行
# ============================================
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

# 每天凌晨 3:00 執行完整流程（全部頻道）
0 3 * * * /home/openclaw/Projects/knowledge-pipeline/scripts/run_pipeline.sh >> /home/openclaw/Projects/knowledge-pipeline/logs/cron.log 2>&1
